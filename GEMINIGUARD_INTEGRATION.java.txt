package com.gemini.guard;

import android.app.BroadcastOptions;
import android.content.Context;
import android.content.Intent;
import android.os.Build;

public final class XiaoAiBackTrigger {
    private XiaoAiBackTrigger() {}

    /** Call immediately AFTER your existing "open Gemini" operation.
     * The Xiao AI system overlay should still own input focus when the ADB BACK arrives.
     */
    public static void send(Context context) {
        Intent intent = new Intent("com.ldd.xiaoaitogemini.RUN_BACK");
        intent.setClassName("com.ldd.xiaoaitogemini", "com.ldd.xiaoaitogemini.RelayReceiver");
        if (Build.VERSION.SDK_INT >= 34) {
            BroadcastOptions options = BroadcastOptions.makeBasic();
            options.setShareIdentityEnabled(true); // needed for receiver to authenticate sender UID
            context.sendBroadcast(intent, null, options.toBundle());
        }
    }
}
